Project 2: GitHub Repo Finder

Search GitHub repositories by keyword and present the top results with star counts, descriptions, and links. Save results to a file.

🚀 Project overview

Build a GitHub repository search tool. The user enters a search query; your program fetches the top results sorted by stars and presents them cleanly. Optionally save results to a CSV or JSON file.

Skills used: Authentication · GET · search params · pagination · JSON · file I/O

Complete Code

import os, json, requests

token = os.environ.get("GITHUB_TOKEN")
headers = {"Authorization": f"Bearer {token}"} if token else {}
BASE = "https://api.github.com"

def search_repos(query, sort="stars", limit=10):
    per_page = min(limit, 30)
    try:
        response = requests.get(
            f"{BASE}/search/repositories",
            headers=headers,
            params={"q": query, "sort": sort,
                    "order": "desc", "per_page": per_page},
            timeout=10
        )
        response.raise_for_status()
    except requests.exceptions.RequestException as e:
        print(f"Error: {e}")
        return []

    return response.json().get("items", [])[:limit]

def display_repos(repos, query):
    print(f"\nTop {len(repos)} results for \"{query}\"")
    print("═" * 42)
    for i, repo in enumerate(repos, 1):
        desc = repo.get("description") or "No description"
        print(f"\n{i}. {repo['full_name']}")
        print(f"   ⭐ {repo['stargazers_count']:,}  🍴 {repo['forks_count']:,}")
        print(f"   {desc}")
        print(f"   {repo['html_url']}")

def save_results(repos, filename="results.json"):
    output = [{
        "name": r["full_name"],
        "stars": r["stargazers_count"],
        "description": r.get("description"),
        "url": r["html_url"],
    } for r in repos]
    with open(filename, "w") as f:
        json.dump(output, f, indent=2)
    print(f"Saved to {filename}")

def main():
    query = input("Search GitHub repos: ").strip()
    sort = input("Sort by (stars/forks/updated) [stars]: ").strip() or "stars"
    limit = int(input("Max results [10]: ").strip() or 10)

    repos = search_repos(query, sort, limit)
    if not repos:
        print("No results."); return

    display_repos(repos, query)

    if input("\nSave results to file? (y/n): ").lower() == "y":
        save_results(repos)

if __name__ == "__main__":
    main()
