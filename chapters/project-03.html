Project 3: News Aggregator

The final project. Fetch live news headlines on any topic, display them cleanly, and save a running log to a JSON file — a complete real-world automation.

🚀 Project overview

Build a news aggregator that fetches top headlines for any topic using the NewsAPI. This project adds scheduled/repeated requests, saves results to a JSON log, and demonstrates a complete real-world automation pattern.

Skills used: API keys · GET · params · JSON · file I/O · datetime · all error handling patterns

💡 Setup: NewsAPI key

Register free at newsapi.org. Free tier: 100 requests/day. Store as export NEWS_API_KEY="your_key".

Complete Code

import os, json, requests
from datetime import datetime

BASE = "https://newsapi.org/v2/everything"

def fetch_news(topic, page_size=5):
    api_key = os.environ.get("NEWS_API_KEY")
    if not api_key:
        raise EnvironmentError("Set NEWS_API_KEY")

    try:
        response = requests.get(
            BASE,
            params={
                "q": topic,
                "sortBy": "publishedAt",
                "pageSize": page_size,
                "language": "en",
                "apiKey": api_key,
            },
            timeout=10
        )
        response.raise_for_status()
    except requests.exceptions.HTTPError:
        err = response.json().get("message", "HTTP error")
        print(f"API error: {err}"); return []
    except requests.exceptions.RequestException as e:
        print(f"Connection error: {e}"); return []

    data = response.json()
    return data.get("articles", [])

def display_articles(articles, topic):
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    print(f"\n📰 Top {len(articles)} headlines for \"{topic}\"")
    print(now)
    print("═" * 50)
    for i, a in enumerate(articles, 1):
        source = a.get("source", {}).get("name", "Unknown")
        pub = (a.get("publishedAt") or "")[:10]
        title = a.get("title", "No title")
        url   = a.get("url", "")
        print(f"\n{i}. {title}")
        print(f"   Source: {source} | {pub}")
        print(f"   {url}")

def save_to_log(articles, topic, filename="news_log.json"):
    # Load existing log or start fresh
    try:
        with open(filename) as f:
            log = json.load(f)
    except FileNotFoundError:
        log = []

    for a in articles:
        log.append({
            "topic": topic,
            "fetched_at": datetime.now().isoformat(),
            "title": a.get("title"),
            "source": a.get("source", {}).get("name"),
            "url": a.get("url"),
        })

    with open(filename, "w") as f:
        json.dump(log, f, indent=2)

    print(f"\nSaved to {filename} ({len(log)} total entries)")

def main():
    topic = input("Topic: ").strip()
    max_a = int(input("Max articles [5]: ").strip() or 5)

    articles = fetch_news(topic, max_a)
    if articles:
        display_articles(articles, topic)
        save_to_log(articles, topic)
    else:
        print("No articles found.")

if __name__ == "__main__":
    main()
