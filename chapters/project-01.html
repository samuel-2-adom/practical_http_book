Project 1: Weather CLI

Your first complete project. A real command-line program that fetches live weather for any city using the OpenWeatherMap API.

🚀 Project overview

Build a command-line weather application that fetches real-time conditions for any city. This project combines everything from Part I–III: URLs, auth, GET requests, JSON parsing, and error handling.

Skills used: requests.get() · params= · headers= · response.json() · os.environ · error handling · f-strings

Step 1 — Setup

# weather.py
import os
import requests

BASE_URL = "https://api.openweathermap.org/data/2.5/weather"

def get_api_key():
    key = os.environ.get("OWM_API_KEY")
    if not key:
        raise EnvironmentError("Set OWM_API_KEY environment variable first.")
    return key

Step 2 — Fetch Function

def fetch_weather(city, api_key, units="metric"):
    try:
        response = requests.get(
            BASE_URL,
            params={"q": city, "appid": api_key, "units": units},
            timeout=10
        )
    except requests.exceptions.ConnectionError:
        print("No internet connection.")
        return None
    except requests.exceptions.Timeout:
        print("Request timed out.")
        return None

    if response.status_code == 404:
        print(f"City '{city}' not found. Check spelling.")
        return None
    elif response.status_code == 401:
        print("Invalid API key.")
        return None
    elif not response.ok:
        print(f"Error {response.status_code}")
        return None

    return response.json()

Step 3 — Display Function

def display_weather(data, units):
    sym = "°C" if units == "metric" else "°F"
    city    = f"{data['name']}, {data['sys']['country']}"
    temp    = data["main"]["temp"]
    feels   = data["main"]["feels_like"]
    hum     = data["main"]["humidity"]
    cond    = data["weather"][0]["description"]
    wind    = data["wind"]["speed"]
    vis_m   = data.get("visibility", 0)
    vis_km  = round(vis_m / 1000, 1)

    print(f"\n🌤  Weather in {city}")
    print("─" * 30)
    print(f"Temperature : {temp}{sym}  (feels like {feels}{sym})")
    print(f"Conditions  : {cond}")
    print(f"Humidity    : {hum}%")
    print(f"Wind        : {wind} m/s")
    print(f"Visibility  : {vis_km} km")

Step 4 — Main Program

def main():
    try:
        api_key = get_api_key()
    except EnvironmentError as e:
        print(e); return

    city = input("City: ").strip()
    unit_input = input("Units (C/F) [C]: ").strip().upper()
    units = "imperial" if unit_input == "F" else "metric"

    data = fetch_weather(city, api_key, units)
    if data:
        display_weather(data, units)

if __name__ == "__main__":
    main()
